# Production dependencies.
#
# Every package here is on PyPI and installs on any host. The previous version
# of this file was the Emergent platform image manifest: it named
# `emergentintegrations` (never published) and fetched litellm from
# customer-assets.emergentagent.com, so `pip install -r requirements.txt`
# succeeded only inside that image and failed at build time everywhere else.
#
# Test and lint tooling lives in requirements-dev.txt so it stays out of the
# production container.
#
#   pip install -r requirements.txt

# --- Web framework ---
fastapi==0.110.1
uvicorn[standard]==0.25.0
starlette==0.37.2
# Multipart form parsing, required by FastAPI for the file-upload endpoints.
python-multipart==0.0.9

# --- Data / validation ---
pydantic==2.9.2
# Backs pydantic's EmailStr, used on every account and invitation model.
email-validator==2.2.0
python-dotenv==1.0.1

# --- Database ---
motor==3.5.1
pymongo==4.8.0
# Required for MongoDB Atlas: its connection strings use the mongodb+srv://
# scheme, which resolves through a DNS SRV lookup. Without this, pymongo fails
# with an error that does not mention DNS and reads like a wrong password.
dnspython==2.6.1

# --- Auth ---
PyJWT==2.9.0
passlib==1.7.4
# passlib's hashing backend. passlib does not declare it, so it is pinned here
# or password verification fails at runtime rather than at install.
bcrypt==4.1.3

# --- HTTP client ---
# httpx only. `requests` is synchronous and stalls the event loop when called
# from an async handler; tests/test_provider_decoupling.py enforces this.
httpx==0.27.2

# --- PDF generation (imported at module scope by pdf_export.py) ---
reportlab==4.2.5
pypdf==5.1.0
pillow==10.4.0

# --- Scheduler ---
APScheduler==3.10.4

# --- Email (providers/email.py) ---
resend==2.4.0
